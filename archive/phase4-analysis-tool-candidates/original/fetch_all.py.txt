    _em_adapter = HTTPAdapter(max_retries=Retry(
        total=3, connect=3, backoff_factor=0.6,
        status_forcelist=[429, 500, 502, 503, 504], allowed_methods=["GET"]))
    EM_SESSION.mount("https://", _em_adapter)
    EM_SESSION.mount("http://", _em_adapter)

_em_last_call = [0.0]

GAPS = []

def em_get(url, params=None, headers=None, timeout=15, **kw):
    wait = EM_MIN_INTERVAL - (time.time() - _em_last_call[0])
    if wait > 0:
        time.sleep(wait + random.uniform(0.1, 0.5))
    # push2 502 风控时走 push2delay 备用主机（延迟行情，研究场景可接受）
    try:
        resp = EM_SESSION.get(url, params=params, headers=headers, timeout=timeout, **kw)
        _em_last_call[0] = time.time()
        return resp
    except Exception as e:
        _em_last_call[0] = time.time()
        alt = url.replace("://push2.", "://push2delay.")
        if alt != url:
            time.sleep(2)
            try:
                print(f"[RETRY-DELAY] {url.split('/')[2]} -> push2delay")
                return EM_SESSION.get(alt, params=params, headers=headers, timeout=timeout, **kw)
            except Exception as e2:
                GAPS.append({"url": url, "error": str(e), "alt_error": str(e2)})
                raise
        GAPS.append({"url": url, "error": str(e)})
        raise

def eastmoney_datacenter(report_name, columns="ALL", filter_str="", page_size=50,
                         sort_columns="", sort_types="-1"):
    params = {"reportName": report_name, "columns": columns, "filter": filter_str,
              "pageNumber": "1", "pageSize": str(page_size), "sortColumns": sort_columns,
              "sortTypes": sort_types, "source": "WEB", "client": "WEB"}
    r = em_get(DATACENTER_URL, params=params, timeout=15)
    d = r.json()
    if d.get("result") and d["result"].get("data"):
        return d["result"]["data"]
    return []

def baidu_kline(code, start_time="20250101"):
    url = "https://finance.pae.baidu.com/selfselect/getstockquotation"
    params = {"all": "1", "isIndex": "false", "isBk": "false", "isBlock": "false",
              "isFutures": "false", "isStock": "true", "newFormat": "1",
              "group": "quotation_kline_ab", "finClientType": "pc",
              "code": code, "start_time": start_time, "ktype": "1"}
    headers = {"User-Agent": "Mozilla/5.0",
               "Accept": "application/vnd.finance-web.v1+json",
               "Origin": "https://gushitong.baidu.com", "Referer": "https://gushitong.baidu.com/"}
    r = requests.get(url, params=params, headers=headers, timeout=15)
    md = r.json().get("Result", {}).get("newMarketData", {})
    keys = md.get("keys", [])
    rows = md.get("marketData", "").split(";")
    parsed = []
    kidx = {k: i for i, k in enumerate(keys)}
    for row in rows:
        parts = row.split(",")
        if len(parts) < len(keys):
            continue
        parsed.append({"time": parts[kidx.get("time", 0)],
                       "open": parts[kidx.get("open", 1)], "close": parts[kidx.get("close", 2)],
                       "high": parts[kidx.get("high", 3)], "low": parts[kidx.get("low", 4)],
                       "volume": parts[kidx.get("volume", 5)]})
    return parsed
kl = baidu_kline(CODE)

for i, k in enumerate(kl):
    if i >= 59:

    if i >= 119:
        k["ma120"] = round(sum(closes[i-119:i+1]) / 120, 2)
save("baidu_kline", kl)

def sina_fin(code, rtype, num=9):
    url = "https://quotes.sina.cn/cn/api/openapi.php/CompanyFinanceService.getFinanceReport2022"
    params = {"paperCode": f"sz{code}", "source": rtype, "type": "0", "page": "1", "num": str(num)}
    r = requests.get(url, params=params, headers={"User-Agent": UA}, timeout=15)
    rl = r.json().get("result", {}).get("data", {}).get("report_list", {}) or {}
    rows = []
    for period in sorted(rl.keys(), reverse=True)[:num]:
        obj = rl[period]
        rec = {"报告期": f"{period[:4]}-{period[4:6]}-{period[6:8]}"}
        for it in obj.get("data", []) or []:
            t = it.get("item_title", "")
            if t and it.get("item_value") is not None:
                rec[t] = it.get("item_value")
        rows.append(rec)
    return rows

def cninfo_orgid(code):
    url = "http://www.cninfo.com.cn/new/data/szse_stock.json"
    r = requests.get(url, headers={"User-Agent": UA}, timeout=20)
    for st in r.json().get("stockList", []):
        if st.get("code") == code:
            return st.get("orgId")
    return None
